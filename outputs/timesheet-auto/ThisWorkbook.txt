Option Explicit

' Paste this entire file into the ThisWorkbook code module.
' Save the workbook as Excel Macro-Enabled Workbook (*.xlsm).
' Clicking the normal + tab creates the next month automatically.

Private Sub Workbook_NewSheet(ByVal Sh As Object)
    Dim source As Worksheet, ws As Worksheet
    Dim latest As Date, candidate As Date, nextMonth As Date
    Dim yearNumber As Long, monthNumber As Long, r As Long, col As Long
    Dim rule As FormatCondition
    Dim oldEvents As Boolean, oldScreen As Boolean
    Dim errorText As String

    If Not TypeOf Sh Is Worksheet Then Exit Sub
    ' Do not overwrite a sheet inserted by copying an existing sheet.
    If Application.CountA(Sh.UsedRange) <> 0 Then Exit Sub

    For Each ws In Me.Worksheets
        If ws.Name <> Sh.Name Then
            If ReadMonthName(ws.Name, yearNumber, monthNumber) Then
                candidate = DateSerial(yearNumber, monthNumber, 1)
                If candidate > latest Then
                    latest = candidate
                    Set source = ws
                End If
            End If
        End If
    Next ws
    If source Is Nothing Then Exit Sub

    ' This file's September sheet has Year/Month at A5/B5 and header row 7.
    If Not IsNumeric(source.Range("A5").Value2) Then Exit Sub
    If Not IsNumeric(source.Range("B5").Value2) Then Exit Sub
    If Len(CStr(source.Range("C7").Value2)) = 0 Then Exit Sub

    oldEvents = Application.EnableEvents
    oldScreen = Application.ScreenUpdating
    On Error GoTo Failed
    Application.EnableEvents = False
    Application.ScreenUpdating = False

    nextMonth = DateAdd("m", 1, latest)
    source.Range("A1:K38").Copy Destination:=Sh.Range("A1")
    source.Range("A37:K37").Copy Destination:=Sh.Range("A38")
    Application.CutCopyMode = False
    For col = 1 To 11
        Sh.Columns(col).ColumnWidth = source.Columns(col).ColumnWidth
    Next col
    For r = 1 To 38
        Sh.Rows(r).RowHeight = source.Rows(r).RowHeight
    Next r
    Sh.Rows(38).RowHeight = source.Rows(37).RowHeight

    Sh.Range("A5").Value2 = Year(nextMonth)
    Sh.Range("B5").Value2 = Month(nextMonth)
    ' New month starts with empty attendance and project entries.
    Sh.Range("C8:K38").ClearContents
    Sh.Range("A8:K38").FormatConditions.Delete
    Sh.Range("A8:K38").Interior.Color = RGB(255, 255, 255)
    Sh.Range("A8:K38").Font.Color = RGB(0, 0, 0)
    Sh.Range("F8:F38").Interior.Color = RGB(244, 177, 131)

    For r = 8 To 38
        Sh.Cells(r, 1).Formula = "=IF(ROW()-7<=DAY(DATE($A$5,$B$5+1,0)),DATE($A$5,$B$5,ROW()-7),"""")"
        Sh.Cells(r, 2).Formula = "=IF(A" & r & "="""","""",TEXT(A" & r & ",""ddd""))"
        Sh.Cells(r, 6).Formula = "=IF(OR(A" & r & "="""",COUNT(C" & r & ":D" & r & ")<2),"""",MOD(D" & r & "-C" & r & ",1)-E" & r & ")"
    Next r
    Sh.Range("A8:A38").NumberFormat = "yyyy""" & ChrW(&H5E74) & """m""" & ChrW(&H6708) & """d""" & ChrW(&H65E5) & """"
    Sh.Range("C8:G38").NumberFormat = "h:mm"
    Sh.Range("J8:J38").NumberFormat = "h:mm"

    Set rule = Sh.Range("A8:K38").FormatConditions.Add( _
        Type:=xlExpression, Formula1:="=AND(ISNUMBER($A8),WEEKDAY($A8,2)>5)")
    rule.Interior.Color = RGB(191, 191, 191)
    rule.Font.Color = RGB(0, 0, 0)
    rule.StopIfTrue = True
    Set rule = Sh.Range("A8:K38").FormatConditions.Add( _
        Type:=xlExpression, Formula1:="=$A8=""""")
    rule.Interior.Color = RGB(255, 255, 255)
    rule.StopIfTrue = True

    Sh.Name = CStr(Year(nextMonth)) & ChrW(&H5E74) & Format$(Month(nextMonth), "00")
    If Sh.Index <> Me.Sheets.Count Then Sh.Move After:=Me.Sheets(Me.Sheets.Count)
    Sh.Calculate
    Sh.Activate
    Sh.Range("C8").Select
Finished:
    Application.CutCopyMode = False
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreen
    Exit Sub
Failed:
    errorText = Err.Description
    Application.CutCopyMode = False
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreen
    MsgBox "Could not finish creating the month: " & errorText, vbExclamation
End Sub

Private Function ReadMonthName(ByVal tabName As String, ByRef y As Long, ByRef m As Long) As Boolean
    If Len(tabName) <> 7 Then Exit Function
    If Mid$(tabName, 5, 1) <> ChrW(&H5E74) Then Exit Function
    If Not Left$(tabName, 4) Like "####" Then Exit Function
    If Not Right$(tabName, 2) Like "##" Then Exit Function
    y = CLng(Left$(tabName, 4))
    m = CLng(Right$(tabName, 2))
    ReadMonthName = (y >= 1900 And y <= 9998 And m >= 1 And m <= 12)
End Function
